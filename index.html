import numpy
import scipy.special

class neuralNetwork:
  def __init__(self, inputnodes, hiddennodes, outputnodes, learningrate):
    #set nuumber of nodes in each layer
    self.inodes = inputnodes
    self.hnodes = hiddennodes
    self.onodes = outputnodes

    # learning rate
    self.lr = learningrate

    # link weight matrices, wih and who
    self.wih = numpy.random.normal(0.0, pow(self.hnodes, -0.5), (self.hnodes, self.inodes))
    self.who = numpy.random.normal(0.0, pow(self.onodes, -0.5), (self.onodes, self.hnodes))

    # activation function is the sigmoid function
    self.activation_function = lambda x: scipy.special.expit(x)
    pass

  # train the neural network
  def train(self, inputs_list, targets_list):
    # convert inputs list to 2d array
    inputs = numpy.array(inputs_list, ndmin=2).T
    targets = numpy.array(targets_list, ndmin=2).T

    # calculate signals into hidden layer
    hidden_inputs = numpy.dot(self.wih, inputs)

    # calculate the signals emerging from hidden layer
    hidden_outputs = self.activation_function(hidden_inputs)

    # calculate signals into final output layer
    final_inputs = numpy.dot(self.who, hidden_outputs)

    # calculate the signals emerging from final output layer
    final_outputs = self.activation_function(final_inputs)

    output_errors = targets - final_outputs
    hidden_errors = numpy.dot(self.who.T, output_errors)
    self.who += self.lr * numpy.dot((output_errors * final_outputs * (1.0 - final_outputs)), numpy.transpose(hidden_outputs))

    self.wih += self.lr * numpy.dot((hidden_errors * hidden_outputs * (1.0 - hidden_outputs)), numpy.transpose(inputs))

    pass

  def query(self, inputs_list):
    inputs = numpy.array(inputs_list, ndmin=2).T
    # calculate signals into hidden layer
    hidden_inputs = numpy.dot(self.wih, inputs)
    # calculate the signals emerging from hidden layer
    hidden_outputs = self.activation_function(hidden_inputs)
    # calculate signals into final output layer
    final_inputs = numpy.dot(self.who, hidden_outputs)
    # calculate the signals emerging from final output layer
    final_outputs = self.activation_function(final_inputs)
    return final_outputs

# DF1: [EarningsCall ID, CompanyName, CallDate, EarningsCallText]
# DF2: [SpeakerTurnID(New round unique ID), EarningsCallID,SpeakerName, PreparedOrQA, Text]

FOOTER_START = "this article is a transcript of this conference call"

QA_CUES = re.compile(
    r"question[- ]and[- ]answer"
    r"|q\s?&\s?a"
    r"|first question"
    r"|operator instructions"
    r"|open (up )?the (line|lines|floor|call) (up )?(for|to) (your )?questions"
    r"|(begin|start|take|conduct)\w* (the |our |a )?question"
    r"|ready to take questions",
    re.IGNORECASE,
)
# Phrases that mention Q&A but say there is none (e.g. Hain Celestial)
NO_QA = re.compile(r"\b(not|no)\b[^.]{0,40}\b(q\s?&\s?a|questions?)\b", re.IGNORECASE)

def find_qa_start(turns):
    """Return the index of the first Q&A turn, or None if the call has no Q&A.

    1. Prefer the first Operator turn (after the opening) that announces Q&A.
       This avoids switching early when an executive says something like
       "before we move to questions" at the end of the prepared remarks.
    2. If the operator never announces it (calls moderated by an IR person),
       fall back to the first turn by anyone that announces Q&A.
    """
    for i, t in enumerate(turns):
        if i >= 1 and t["SpeakerName"] == "Operator" and QA_CUES.search(t["Text"]) \
                and not NO_QA.search(t["Text"]):
            return i
    for i, t in enumerate(turns):
        if i >= 2 and QA_CUES.search(t["Text"]) and not NO_QA.search(t["Text"]):
            return i
    return None

# 2. Keep only Prepared remarks and management RESPONSES in Q&A
#    - drop all Operator turns
#    - analyst = the speaker right after an Operator turn in Q&A who did not
#      speak in the prepared remarks (so a CEO giving closing remarks after
#      the operator is not mistaken for an analyst); drop all their turns
#    - drop "question routing" turns from IR moderators (e.g. Oil-Dri, New Horizon)
#    - drop closing remarks after the operator ends the Q&A (not answers to questions)
import re

CLOSING = re.compile(
    r"no (further|more|additional) questions"
    r"|(concludes|conclude|end of|ending|wrap up)\w* (our|the|today's|this)?\s*(question|q\s?&\s?a)"
    r"|(turn|hand)\w* (the call|it|the floor|the conference|things) (over )?back (over )?to",  # needs "back"
    re.IGNORECASE)

ROUTING = re.compile(
    r"\b(next|first|final|last|another|following)\s+(analyst\s+)?question\b"
    r"|\bquestion\s+(comes|is coming|is|has come)\s+from\b"
    r"|\bquestions?\s+from\s+(the line of|reddit)\b",
    re.IGNORECASE)

def assign_roles(call):
    call = call.sort_values("SpeakerTurnID")
    prepared_speakers = set(call.loc[call.Section == "Prepared", "SpeakerName"]) - {"Operator"}

    # find analysts for this call
    analysts = set()
    prev = None
    for _, r in call.iterrows():
        if (r.Section == "QnA" and prev == "Operator"
                and r.SpeakerName != "Operator"
                and r.SpeakerName not in prepared_speakers):
            analysts.add(r.SpeakerName)
        prev = r.SpeakerName

    roles = []
    qa_over = False
    for _, r in call.iterrows():
        if r.SpeakerName == "Operator":
            roles.append("Operator")
            if r.Section == "QnA" and CLOSING.search(r.Text):
                qa_over = True
        elif r.Section == "QnA" and qa_over:
            roles.append("Closing")
        elif r.Section == "Prepared":
            roles.append("Prepared")
        elif r.SpeakerName in analysts:
            roles.append("Analyst")
        elif ROUTING.search(r.Text):
            roles.append("Moderator")
        else:
            roles.append("Response")
    return pd.Series(roles, index=call.index)

# Training data: one row per SENTENCE (the webpage classifies a single sentence)

def split_sentences(text):
    return [s.strip() for s in re.split(r"(?<=[.!?])\s+", text) if s.strip()]

def tokenize(text):
    return re.findall(r"[a-z']+", text.lower())

## Step 3: Naive Bayes model (multinomial, Laplace smoothing, with class priors)
CLASSES = ["Prepared", "QnA"]

def train_nb(df):
    counts = {c: Counter() for c in CLASSES}
    n_docs = Counter(df.Label)
    for _, r in df.iterrows():
        counts[r.Label].update(tokenize(r.Sentence))

    vocab = set(counts["Prepared"]) | set(counts["QnA"])
    V = len(vocab)
    model = {"V": V, "prior": {}, "logprob": {}, "unseen": {}, "total_words": {}}
    for c in CLASSES:
        N = sum(counts[c].values())
        model["prior"][c] = math.log(n_docs[c] / len(df))
        model["unseen"][c] = math.log(1 / (N + V))          # word in vocab but never seen in this class
        model["logprob"][c] = {w: math.log((counts[c][w] + 1) / (N + V)) for w in vocab}
        model["total_words"][c] = N
    model["counts"] = counts
    return model

def score_sentence(model, sentence):
    """Returns the score for each class and each word's contribution."""
    scores = {c: model["prior"][c] for c in CLASSES}
    words = []
    for w in tokenize(sentence):
        if w not in model["logprob"]["Prepared"]:
            words.append((w, None))        # not in vocabulary: ignored (same for both classes)
            continue
        for c in CLASSES:
            scores[c] += model["logprob"][c][w]
        words.append((w, model["logprob"]["Prepared"][w] - model["logprob"]["QnA"][w]))
    pred = max(scores, key=scores.get)
    return pred, scores, words
